#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 定义顺序表结构
#define MAX_SIZE 100
typedef int ElemType;
typedef struct {
    ElemType data[MAX_SIZE];
    int length;
} SqList;

// 顺序表就地逆置
void reverseSqList(SqList *L) {
    int left = 0;
    int right = L->length - 1;
    while (left < right) {
        ElemType temp = L->data[left];
        L->data[left] = L->data[right];
        L->data[right] = temp;
        left++;
        right--;
    }
}

// 定义单链表结构
typedef struct LNode {
    ElemType data;
    struct LNode *next;
} LNode, *LinkList;

// 合并两个递增有序单链表为递减有序单链表
LinkList mergeAndReverse(LinkList A, LinkList B) {
    LinkList C = NULL;
    LinkList tail = NULL;
    LinkList current;
    while (A != NULL || B != NULL) {
        if (A == NULL) {
            current = B;
            B = B->next;
        } else if (B == NULL) {
            current = A;
            A = A->next;
        } else if (A->data < B->data) {
            current = A;
            A = A->next;
        } else {
            current = B;
            B = B->next;
        }
        current->next = C;
        C = current;
    }
    return C;
}

// 求两个递增有序顺序表的交集
void intersectionOfSqLists(SqList A, SqList B, SqList *C) {
    int i = 0, j = 0, k = 0;
    while (i < A.length && j < B.length) {
        if (A.data[i] == B.data[j]) {
            C->data[k++] = A.data[i];
            i++;
            j++;
        } else if (A.data[i] < B.data[j]) {
            i++;
        } else {
            j++;
        }
    }
    C->length = k;
}

// 删除A表中既在B表又在C表中出现的元素
void deleteSharedItems(SqList *A, SqList B, SqList C) {
    int i, j, k;
    for (i = 0; i < A->length;) {
        int foundInB = 0, foundInC = 0;
        for (j = 0; j < B.length; j++) {
            if (A->data[i] == B.data[j]) {
                foundInB = 1;
                break;
            }
        }
        for (k = 0; k < C.length; k++) {
            if (A->data[i] == C.data[k]) {
                foundInC = 1;
                break;
            }
        }
        if (foundInB && foundInC) {
            for (j = i; j < A->length - 1; j++) {
                A->data[j] = A->data[j + 1];
            }
            A->length--;
        } else {
            i++;
        }
    }
}

// 删除单向循环链表中某个结点的前驱结点
void deletePredecessor(LNode *S) {
    if (S == NULL || S->next == S) {
        return;
    }
    LNode *prev = S;
    LNode *current = S->next;
    while (current->next != S) {
        prev = current;
        current = current->next;
    }
    prev->next = current->next;
    free(current);
}

// 判断输入的字符序列是否为回文
int isPalindrome() {
    char stack[MAX_SIZE];
    int top = -1;
    char ch;
    while ((ch = getchar()) != '@') {
        stack[++top] = ch;
    }
    while ((ch = getchar()) != '@') {
        if (top < 0 || ch != stack[top--]) {
            return 0;
        }
    }
    return top == -1;
}

// 递归求F(n)
int recursiveF(int n) {
    if (n == 0) {
        return 1;
    }
    return n * recursiveF(n / 2);
}

// 消除递归求F(n)
int iterativeF(int n) {
    int result = 1;
    while (n > 0) {
        result *= n;
        n /= 2;
    }
    return result;
}

// 火车车厢调度
void trainScheduling(char *cars, int n) {
    char stack[MAX_SIZE];
    int top = -1;
    int outputIndex = 0;
    char output[MAX_SIZE * 2];
    for (int i = 0; i < n; i++) {
        if (cars[i] == 'S') {
            output[outputIndex++] = 'S';
        } else {
            stack[++top] = cars[i];
        }
    }
    while (top >= 0) {
        output[outputIndex++] = stack[top--];
    }
    output[outputIndex] = '\0';
    printf("调度操作序列: %s\n", output);
}
